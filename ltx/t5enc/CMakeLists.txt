cmake_minimum_required(VERSION 3.18)
project(t5enc CXX C)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${SD_SRC} sd)
add_executable(t5enc t5enc.cpp)
target_include_directories(t5enc PRIVATE ${SD_SRC}/src)
target_link_libraries(t5enc PRIVATE stable-diffusion)
target_compile_definitions(t5enc PRIVATE GGML_MAX_NAME=160)
target_compile_features(t5enc PUBLIC cxx_std_17)
